function calcularEfectividad(tipoAtaque, tiposDefensor) {
    let mult = 1;
    for (let tipo of tiposDefensor) {
        if (EFECTIVIDAD[tipoAtaque] && EFECTIVIDAD[tipoAtaque][tipo] !== undefined) {
            mult *= EFECTIVIDAD[tipoAtaque][tipo];
        }
    }
    return mult;
}

function crearPokemon(data, escalarNivel = 50) {
    const factor = 1 + ((escalarNivel - 5) * 0.02);
    const stats = {
        hp: Math.floor(data.stats.hp * factor),
        atk: Math.floor(data.stats.atk * factor),
        def: Math.floor(data.stats.def * factor),
        atkE: Math.floor(data.stats.atkE * factor),
        defE: Math.floor(data.stats.defE * factor),
        spd: Math.floor(data.stats.spd * factor)
    };
    const movs = data.movs.map(n => ({
        nombre: n,
        ...MOVIMIENTOS[n],
        ppActual: MOVIMIENTOS[n].pp,
        ppMax: MOVIMIENTOS[n].pp
    }));
    return {
        nombre: data.nombre,
        tipos: [...data.tipos],
        stats: stats,
        hpActual: stats.hp,
        hpMax: stats.hp,
        movimientos: movs,
        tier: data.tier || null,
        esJefe: data.esJefe || false,
        hpEscudo: data.hpEscudo || 0,
        hpEscudoMax: data.hpEscudoMax || 0,
        hpVida: data.hpVida || 0,
        hpVidaMax: data.hpVidaMax || 0,
        fase: data.fase || 1,
        enfurecido: false,
        movExclusivo: data.movExclusivo || null,
        turnoFase2: 0
    };
}

function hpBarClass(actual, max) {
    const pct = (actual / max) * 100;

    if (game.equipo.length >= 4) {

function iniciarOleada() {
    mostrarPantalla('battle');

    clearLog();
    
    if (game.oleada % 10 === 0) {

    } else {

function generarEnemigos() {
    const nivel = 5 + (game.oleada * 2);
    let cantidad;
    if (game.oleada <= 3) cantidad = 1 + Math.floor(Math.random() * 2);
    else if (game.oleada <= 10) cantidad = 2 + Math.floor(Math.random() * 2);
    else cantidad = 3 + Math.floor(Math.random() * 2);
    
    game.enemigos = [];
    const esPico = (game.oleada % 5 === 0) && (game.oleada % 10 !== 0);
    
    for (let i = 0; i < cantidad; i++) {
        const tier = seleccionarTier(game.oleada, esPico);
        const pool = POOL_POKEMON[tier];
        const base = pool[Math.floor(Math.random() * pool.length)];
        const enem = crearPokemon(base, nivel);
        enem.tier = tier;
        
        const afijo = seleccionarAfijo(game.oleada, esPico);
        aplicarAfijo(enem, afijo);
        enem.afijo = afijo;
        
        game.enemigos.push(enem);
    }
}

function seleccionarTier(oleada, esPico) {
    let roll = Math.floor(Math.random() * 100);
    if (esPico) roll -= 20;
    
    if (oleada < 5) {
        if (roll < 70) return 'Basico';
        if (roll < 95) return 'Fase2';
        return 'Final';
    } else if (oleada < 15) {
        if (roll < 40) return 'Basico';
        if (roll < 75) return 'Fase2';
        if (roll < 95) return 'Final';
        return 'Pseudo';
    } else {
        if (roll < 20) return 'Fase2';
        if (roll < 55) return 'Final';
        return 'Pseudo';
    }
}

function seleccionarAfijo(oleada, esPico) {
    if (oleada < 3) return 'Ninguno';
    let roll = Math.floor(Math.random() * 100);
    if (esPico) roll -= 30;
    if (roll < 50) return 'Ninguno';
    if (roll < 65) return 'Furioso';
    if (roll < 80) return 'Blindado';
    if (roll < 90) return 'Veloz';
    return 'Reflejante';
}

function aplicarAfijo(enem, afijo) {
    if (afijo === 'Furioso') {
        enem.stats.atk = Math.floor(enem.stats.atk * 1.3);
        enem.stats.def = Math.floor(enem.stats.def * 0.8);
    } else if (afijo === 'Blindado') {
        enem.stats.def = Math.floor(enem.stats.def * 1.5);
        enem.stats.spd = Math.floor(enem.stats.spd * 0.8);
    } else if (afijo === 'Veloz') {
        enem.stats.spd = Math.floor(enem.stats.spd * 1.4);
    }
}

function seleccionarObjetivo(idx) {
    if (!game.seleccionandoMov) return;
    const {pokemonIdx, movIdx} = game.seleccionandoMov;
    ejecutarAtaque(pokemonIdx, movIdx, idx);
    game.seleccionandoMov = null;
}

function ejecutarAtaque(pokemonIdx, movIdx, targetIdx) {
    const p = game.equipo[pokemonIdx];
    const mov = p.movimientos[movIdx];
    const target = game.enemigos[targetIdx];
    
    mov.ppActual--;
    
    const efect = calcularEfectividad(mov.tipo, target.tipos);
    if (efect === 0) {
        addLog(`${p.nombre} usa ${mov.nombre}... ¡No afecta a ${target.nombre}!`, 'info');
        terminarTurnoJugador();
        return;
    }
    
    let dano;
    if (mov.cat === 'Fisico') {
        dano = mov.pot + (p.stats.atk - target.stats.def);
    } else {
        dano = mov.pot + (p.stats.atkE - target.stats.defE);
    }
    dano = Math.max(1, Math.floor(dano * efect));
    
    const escama = game.reliquias.find(r => r === 'Escama Dragón');
    if (escama && p.tipos.includes('Dragon')) {
        dano = Math.floor(dano * 1.15);
    }
    
    target.hpActual = Math.max(0, target.hpActual - dano);
    
    let efectTxt = '';
    if (efect > 1) efectTxt = ' <span style="color:#51cf66;">¡SUPER EFECTIVO!</span>';
    else if (efect < 1) efectTxt = ' <span style="color:#ff6b6b;">(Poco efectivo)</span>';
    
    addLog(`${tipoBadge(mov.tipo)} <strong>${p.nombre}</strong> usa <strong>${mov.nombre}</strong> contra ${target.nombre} → <strong>${dano}</strong> daño${efectTxt}`, 'damage');
    
    if (target.hpActual <= 0) {
        addLog(`💀 ¡${target.nombre} ha sido derrotado!`, 'critical');
    }
    
    renderizarCombate();
    terminarTurnoJugador();
}

function ejecutarAtaqueJefe(pokemonIdx, movIdx) {
    const p = game.equipo[pokemonIdx];
    const mov = p.movimientos[movIdx];
    const j = game.jefe;
    
    mov.ppActual--;
    
    const efect = calcularEfectividad(mov.tipo, j.tipos);
    
    let dano;
    if (mov.cat === 'Fisico') {
        dano = mov.pot + (p.stats.atk - j.stats.def);
    } else {
        dano = mov.pot + (p.stats.atkE - j.stats.defE);
    }
    dano = Math.max(1, Math.floor(dano * efect));
    
    const escama = game.reliquias.find(r => r === 'Escama Dragón');
    if (escama && p.tipos.includes('Dragon')) {
        dano = Math.floor(dano * 1.15);
    }
    
    if (j.fase === 1) {
        const danoEscudo = efect > 1 ? dano * 2 : dano;
        j.hpEscudo = Math.max(0, j.hpEscudo - danoEscudo);
        addLog(`${tipoBadge(mov.tipo)} <strong>${p.nombre}</strong> usa <strong>${mov.nombre}</strong> → <strong>${danoEscudo}</strong> daño al ESCUDO`, 'damage');
        
        if (j.hpEscudo <= 0) {
            j.fase = 2;
            j.enfurecido = true;
            j.stats.atk += 50;
            j.stats.atkE += 50;
            j.stats.spd += 30;
            addLog(` ¡${j.nombre} DESTRUYÓ SU ARMADURA Y ENTRA EN MODO ENFURECIDO! (+ATK, +SPD)`, 'critical');
        }
    } else {
        j.hpVida = Math.max(0, j.hpVida - dano);
        addLog(`${tipoBadge(mov.tipo)} <strong>${p.nombre}</strong> usa <strong>${mov.nombre}</strong> → <strong>${dano}</strong> daño a VIDA REAL`, 'damage');
    }
    
    renderizarCombate();
    terminarTurnoJugador();
}

function terminarTurnoJugador() {
    if (game.jefe) {
        if (game.jefe.fase === 2 && game.jefe.hpVida <= 0) {
            victoriaOleada();
            return;
        }
    } else {
        if (game.enemigos.every(e => e.hpActual <= 0)) {
            victoriaOleada();
            return;
        }
    }
    
    setTimeout(() => turnoEnemigo(), 800);
}

function turnoEnemigo() {
    if (game.reliquias.includes('Restos Eternos')) {
        game.equipo.forEach(p => {
            if (p.hpActual > 0) {
                const cur = Math.floor(p.hpMax * 0.05);
                p.hpActual = Math.min(p.hpMax, p.hpActual + cur);
                addLog(`💚 ${p.nombre} regenera ${cur} HP (Restos)`, 'heal');
            }
        });
    }
    
    if (game.jefe) {
        ataqueJefe();
    } else {
        game.enemigos.filter(e => e.hpActual > 0).forEach(e => {
            const objetivo = game.equipo.find(p => p.hpActual > 0);
            if (objetivo) ataqueEnemigo(e, objetivo);
        });
    }
    
    renderizarCombate();
    
    if (game.equipo.every(p => p.hpActual <= 0)) {
        setTimeout(gameOver, 1000);
        return;
    }
    
    game.turnoActual++;
    setTimeout(esperarTurnoJugador, 800);
}

function ataqueEnemigo(enem, objetivo) {
    const movsDisponibles = enem.movimientos.filter(m => m.ppActual > 0);
    if (movsDisponibles.length === 0) return;
    const mov = movsDisponibles[Math.floor(Math.random() * movsDisponibles.length)];
    mov.ppActual--;
    
    const efect = calcularEfectividad(mov.tipo, objetivo.tipos);
    if (efect === 0) return;
    
    let dano;
    if (mov.cat === 'Fisico') {
        dano = mov.pot + (enem.stats.atk - objetivo.stats.def);
    } else {
        dano = mov.pot + (enem.stats.atkE - objetivo.stats.defE);
    }
    dano = Math.max(1, Math.floor(dano * efect));
    
    objetivo.hpActual = Math.max(0, objetivo.hpActual - dano);
    
    addLog(`👹 <strong>${enem.nombre}</strong> usa <strong>${mov.nombre}</strong> contra ${objetivo.nombre} → <strong>${dano}</strong> daño`, 'damage');
    
    if (objetivo.hpActual <= 0) {
        addLog(`💀 ¡${objetivo.nombre} ha caído!`, 'critical');
    }
}

function ataqueJefe() {
    const j = game.jefe;
    const objetivo = game.equipo.find(p => p.hpActual > 0);
    if (!objetivo) return;
    
    let mov;
    if (j.enfurecido) {
        j.turnoFase2++;
        if (j.turnoFase2 % 3 === 0) {
            mov = j.movExclusivo;
        }
    }
    if (!mov) {
        const disponibles = j.movimientos.filter(m => m.ppActual > 0);
        mov = disponibles.length > 0 ? disponibles[Math.floor(Math.random() * disponibles.length)] : j.movimientos[0];
        mov.ppActual--;
    }
    
    const efect = calcularEfectividad(mov.tipo, objetivo.tipos);
    if (efect === 0) return;
    
    let dano;
    if (mov.cat === 'Fisico') {
        dano = mov.pot + (j.stats.atk - objetivo.stats.def);
    } else {
        dano = mov.pot + (j.stats.atkE - objetivo.stats.defE);
    }
    dano = Math.max(1, Math.floor(dano * efect));
    
    objetivo.hpActual = Math.max(0, objetivo.hpActual - dano);
    
    addLog(`🔥 <strong>${j.nombre}</strong> usa <strong>${mov.nombre}</strong> contra ${objetivo.nombre} → <strong>${dano}</strong> daño`, 'damage');
    
    if (objetivo.hpActual <= 0) {
        addLog(`💀 ¡${objetivo.nombre} ha caído ante el jefe!`, 'critical');
    }
}

function victoriaOleada() {
    setTimeout(() => {
        mostrarPantalla('reward');
        const recompensa = game.oleada * 2 + 5;
        const bonusJefe = game.jefe ? 10 : 0;
        const total = recompensa + bonusJefe;
        game.soles += total;
        
        game.equipo.forEach(p => {
            if (p.hpActual > 0) {
                const cur = Math.floor(p.hpMax * 0.3);
                p.hpActual = Math.min(p.hpMax, p.hpActual + cur);
            }
        });
        
        const cont = document.getElementById('reward-content');
        cont.innerHTML = `
            <div class="info-box">
                <h3 style="color:#ffd700;">¡Oleada ${game.oleada} superada!</h3>
                <p>💰 +${recompensa} Soles ${bonusJefe > 0 ? `+ ${bonusJefe} bonus jefe` : ''}</p>
                <p>💚 Equipo curado un 30%</p>
                <p><strong>Total Soles: ${game.soles} 💰</strong></p>
            </div>
        `;
    }, 1000);
}

function mostrarSeccionTienda(seccion) {
    const cont = document.getElementById('shop-content');
    cont.innerHTML = '';
    
    if (seccion === 'pokemon') {
        cont.innerHTML = '<h3> Pokémon Disponibles</h3>';
        const grid = document.createElement('div');
        grid.className = 'shop-grid';
        
        const opciones = [];
        ['Basico','Fase2','Final','Pseudo'].forEach(tier => {
            const pool = POOL_POKEMON[tier];
            const pick = pool[Math.floor(Math.random() * pool.length)];
            opciones.push({...pick, tier});
        });
        
        opciones.forEach(op => {
            const coste = PRECIOS[op.tier];
            const puede = game.soles >= coste && game.equipo.length < 4;
            const item = document.createElement('div');
            item.className = 'shop-item' + (!puede ? ' cant-afford' : '');
            item.innerHTML = `
                <div style="display:flex; align-items:center; gap:10px;">
                    <div class="sprite">${SPRITES[op.nombre] || '🐾'}</div>
                    <div style="flex:1;">
                        <strong>${op.nombre}</strong> (${op.tier})
                        <div>${op.tipos.map(t => tipoBadge(t)).join('')}</div>
                        <div style="margin-top:5px;"><strong>${coste} 💰</strong></div>
                    </div>
                </div>
            `;
            if (puede) {
                item.onclick = () => comprarPokemon(op, coste);
            }
            grid.appendChild(item);
        });
        
        cont.appendChild(grid);
        
    } else if (seccion === 'tutor') {
        cont.innerHTML = '<h3>📚 Tutor de Movimientos (3 💰)</h3>';
        if (game.soles < 3) {
            cont.innerHTML += '<p style="color:#ff6b6b;">No tienes suficientes Soles</p>';
            return;
        }
        
        game.equipo.forEach((p, idx) => {
            if (p.hpActual <= 0) return;
            const div = document.createElement('div');
            div.style.cssText = 'background:rgba(255,255,255,0.05); padding:10px; margin:10px 0; border-radius:8px;';
            div.innerHTML = `<strong>${p.nombre}</strong>`;
            
            p.movimientos.forEach((m, mIdx) => {
                const btn = document.createElement('button');
                btn.className = 'btn';
                btn.style.fontSize = '0.85em';
                btn.innerHTML = `Reemplazar: ${m.nombre} →`;
                btn.onclick = () => abrirTutor(idx, mIdx);
                div.appendChild(btn);
            });
            
            cont.appendChild(div);
        });
        
    } else if (seccion === 'reliquias') {
        cont.innerHTML = '<h3>🏺 Reliquias</h3>';
        const grid = document.createElement('div');
        grid.className = 'shop-grid';
        
        Object.keys(RELIQUIAS).forEach(nombre => {
            const r = RELIQUIAS[nombre];
            const puede = game.soles >= r.coste;
            const item = document.createElement('div');
            item.className = 'shop-item' + (!puede ? ' cant-afford' : '');
            item.innerHTML = `
                <h4>${nombre}</h4>
                <p style="font-size:0.9em; opacity:0.8;">${r.desc}</p>
                <p style="margin-top:10px;"><strong>${r.coste} 💰</strong></p>
            `;
            if (puede) {
                item.onclick = () => comprarReliquia(nombre);
            }
            grid.appendChild(item);
        });
        
        cont.appendChild(grid);
        
    } else if (seccion === 'curar') {
        cont.innerHTML = '<h3>💊 Curar Equipo (2 💰 por Pokémon)</h3>';
        if (game.soles < 2) {
            cont.innerHTML += '<p style="color:#ff6b6b;">No tienes suficientes Soles</p>';
            return;
        }
        
        game.equipo.forEach((p, idx) => {
            if (p.hpActual < p.hpMax) {
                const btn = document.createElement('button');
                btn.className = 'btn btn-success';
                btn.innerHTML = `💚 Curar a ${p.nombre} (${p.hpActual}/${p.hpMax}) - 2💰`;
                btn.onclick = () => curarPokemon(idx);
                cont.appendChild(btn);
            }
        });
    }
}

function comprarPokemon(op, coste) {
    if (game.soles < coste || game.equipo.length >= 4) return;
    const nuevo = crearPokemon(op);
    nuevo.tier = op.tier;
    game.equipo.push(nuevo);
    game.soles -= coste;
    document.getElementById('shop-soles').textContent = game.soles;
    alert(`¡Has reclutado a ${op.nombre}!`);
    mostrarSeccionTienda('pokemon');
}

function abrirTutor(pokemonIdx, movIdx) {
    const p = game.equipo[pokemonIdx];
    const movsDisponibles = Object.keys(MOVIMIENTOS);
    const opciones = [];
    while (opciones.length < 4) {
        const pick = movsDisponibles[Math.floor(Math.random() * movsDisponibles.length)];
        if (!opciones.includes(pick)) opciones.push(pick);
    }
    
    let html = `<h4>Reemplazar "${p.movimientos[movIdx].nombre}" de ${p.nombre}:</h4>`;
    opciones.forEach(nombre => {
        const m = MOVIMIENTOS[nombre];
        html += `<button class="btn" style="display:block; margin:5px 0; width:100%;" onclick="ejecutarTutor(${pokemonIdx}, ${movIdx}, '${nombre}')">${tipoBadge(m.tipo)} ${nombre} (Pot:${m.pot})</button>`;
    });
    html += `<button class="btn btn-danger" onclick="mostrarSeccionTienda('tutor')">Cancelar</button>`;
    
    document.getElementById('shop-content').innerHTML = html;
}

function ejecutarTutor(pokemonIdx, movIdx, nuevoMov) {
    if (game.soles < 3) return;
    const p = game.equipo[pokemonIdx];
    const data = MOVIMIENTOS[nuevoMov];
    p.movimientos[movIdx] = {
        nombre: nuevoMov,
        tipo: data.tipo,
        cat: data.cat,
        pot: data.pot,
        ppActual: data.pp,
        ppMax: data.pp
    };
    game.soles -= 3;
    document.getElementById('shop-soles').textContent = game.soles;
    alert(`¡${p.nombre} aprendió ${nuevoMov}!`);
    mostrarSeccionTienda('tutor');
}

function comprarReliquia(nombre) {
    const r = RELIQUIAS[nombre];
    if (game.soles < r.coste) return;
    game.reliquias.push(nombre);
    game.soles -= r.coste;
    document.getElementById('shop-soles').textContent = game.soles;
    alert(`¡Has comprado ${nombre}!`);
    mostrarSeccionTienda('reliquias');
}

function curarPokemon(idx) {
    if (game.soles < 2) return;
    const p = game.equipo[idx];
    p.hpActual = p.hpMax;
    p.movimientos.forEach(m => m.ppActual = m.ppMax);
    game.soles -= 2;
    document.getElementById('shop-soles').textContent = game.soles;
    alert(`¡${p.nombre} ha sido curado completamente!`);
    mostrarSeccionTienda('curar');
}
